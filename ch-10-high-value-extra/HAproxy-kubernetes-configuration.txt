### HA-PROXY CONFIGURATION

# IP - 192.168.1.17 - External (Bridge)
# IP - 10.0.0.199 - Internal (Host Only)

sudo apt update && sudo apt install -y haproxy

vim /etc/haproxy/haproxy.cfg

##add / edit these lines

frontend kubernetes-frontend
    bind 192.168.1.17:6443
    mode tcp
    option tcplog
    default_backend kubernetes-backend

backend kubernetes-backend
    mode tcp
    option tcp-check
    balance roundrobin
    server master-1 10.0.0.101:6443 check
#    server master-2 10.0.0.102:6443 check

## file end here

sudo systemctl restart haproxy
sudo systemctl enable haproxy

#### GATEWAY CONFIGURATION

# Enable packet forwarding

cat <<EOF | sudo tee /etc/sysctl.d/k8s.conf
net.ipv4.ip_forward = 1
EOF

# Apply sysctl params without reboot
sudo sysctl --system

# Configure NAT rules

#Public interface = ens33
#Private interface = ens34 = 10.0.0.199

sudo iptables -t nat -A POSTROUTING -o ens33 -j MASQUERADE

sudo iptables -A FORWARD -i ens33 -o ens34 -m state --state RELATED,ESTABLISHED -j ACCEPT

sudo iptables -A FORWARD -i ens34 -o ens33 -j ACCEPT

sudo apt install iptables-persistent -y

## On Manager after installing packages etc

sudo kubeadm init --control-plane-endpoint "192.168.1.17:6443" --upload-certs --pod-network-cidr=172.17.0.0/16
